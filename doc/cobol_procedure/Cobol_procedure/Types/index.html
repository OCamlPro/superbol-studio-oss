type condition =
  1. | Expr of expr Cobol_common.Srcloc.TYPES.with_loc
    (*

    expression used as a condition

    *)
  2. | Relation of expr Cobol_common.Srcloc.TYPES.with_loc * Cobol_ptree.relop * expr Cobol_common.Srcloc.TYPES.with_loc
    (*
    e <relop> e'
    *)
  3. | ClassCond of expr Cobol_common.Srcloc.TYPES.with_loc * Cobol_ptree.class_
    (*

    class condition

    *)
  4. | SignCond of expr Cobol_common.Srcloc.TYPES.with_loc * Cobol_ptree.signz
    (*
    e POSITIVE/NEGATIVE/ZERO
    *)
  5. | Omitted of expr Cobol_common.Srcloc.TYPES.with_loc
    (*
    c OMITTED
    *)
  6. | Not of condition or_error Cobol_common.Srcloc.TYPES.with_loc
    (*
    NOT c
    *)
  7. | Combined of condition or_error Cobol_common.Srcloc.TYPES.with_loc * Cobol_ptree.logop * condition or_error Cobol_common.Srcloc.TYPES.with_loc
    (*
    c <AND/OR> c'
    *)

Expanded conditions, where all the implied subjects and relational operators of abbreviated conditions are made explicit. Sub-conditions are Error () when they could not be expanded (e.g, missing subject).
