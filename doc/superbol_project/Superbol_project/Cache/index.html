Module Superbol_project.Cache

val storage_location : ?enable_caching:bool -> ?project_layout:Superbol_project__.Project.TYPES.layout -> ?fallback_storage_directory:string -> unit -> (Superbol_project__.Project.TYPES.cache_storage_location, string) Stdlib.result

storage_location ~enable_caching ~project_layout ~fallback_storage_directory () determines if and where project data is going to be cached.

  • fallback_storage_directory, when provided (and if enable_caching holds), is the name of a directory that is used to store a cache whenever project_layout does not provide a per-project storage directory (i.e, project_layout.relative_work_dirname = None).
val default_storage_location : Superbol_project__.Project.TYPES.cache_storage_location
val for_ : rootdir:string -> storage_location:Superbol_project__.Project.TYPES.cache_storage_location -> (string * Superbol_cache.Cache.t) option

for_ ~rootdir ~storage_location returns the cache (along with the directory it lives in) for a project at rootdir and given a storage_location. Returns None if storage_location does not permit caching.
