Module Cobol_preproc.Src_options

Options that drive how source text is read (before preprocessing).

type t = {
  1. tab_stops : int list;
    (*

    Tab stops used to compute visual columns (see Src_format.next_tab_stop).

    *)
  2. mf_comments : bool;
    (*

    Recognize lines with an asterisk in the first column as comments in FIXED, COBOL85, and VARIABLE source formats (cf GnuCOBOL's -fmfcomment).

    *)
}
val default : t

Default tab stops, and no MF-style comment lines.
