type source_format_id =
  1. | SFFree
  2. | SFFixed
  3. | SFCOBOL85
  4. | SFVariable
  5. | SFXOpen
  6. | SFxCard
  7. | SFCRT
  8. | SFTrm
  9. | SFCOBOLX
type file_kind =
  1. | Regular_file
  2. | Directory
  3. | Symbolic_link
    (*

    only reported when links are not followed

    *)
  4. | Other_file
type file_stat = {
  1. file_kind : file_kind;
  2. file_mtime : float;
    (*

    last modification time, in seconds

    *)
  3. file_size : int;
}

What platform.stat_file gives about a file.

type platform = {
  1. mutable verbosity : int;
  2. mutable eprintf : 'a. ('a, Stdlib.out_channel, unit) Stdlib.format -> 'a;
  3. mutable error : 'a. ('a, Stdlib.Format.formatter, unit) Stdlib.format -> 'a;
  4. mutable getcwd : unit -> string;
    (*

    now () is the current time, in seconds since the epoch.

    *)
  5. mutable now : unit -> float;
  6. mutable read_text_file : string -> string;
    (*

    read_text_file_in_path ~search_path filename gives the path and contents of the first file named filename in search_path (or of filename itself if it is not implicit); None if there is none. Raises Sys_error if a file is found but cannot be read. Looking up and reading at once ensures the contents are those of the path given.

    A platform that overrides read_text_file should override this field too (see FS.read_text_file_in_path).

    *)
  7. mutable read_text_file_in_path : search_path:string list -> string -> (string * string) option;
  8. mutable list_dir : string -> string list;
    (*

    stat_file ?follow path describes path, following symbolic links unless follow is false (defaults to true); None if path cannot be reached.

    *)
  9. mutable stat_file : ?follow:bool -> string -> file_stat option;
  10. mutable with_stdin : 'a. f:(Stdlib.in_channel -> 'a) -> 'a;
  11. mutable peek_channel_prefix : Stdlib.in_channel -> len:int -> string;
  12. mutable getenv_opt : string -> string option;
  13. mutable mk_temp_dir : ?mode:int -> ?dir:string -> string -> string;
  14. mutable remove_dir : ?all:bool -> string -> unit;
  15. mutable autodetect_format : source_contents:string -> filename:string -> source_format_id;
    (*

    is_source_file ~filename indicates whether filename may hold COBOL programs (as opposed to only copybooks), judging from its name alone. Used to find the programs of a project.

    *)
  16. mutable is_source_file : filename:string -> bool;
    (*

    is_hidden_file ~filename indicates whether filename (a file or a directory) is hidden, and should be skipped when looking for source files.

    *)
  17. mutable is_hidden_file : filename:string -> bool;
  18. mutable find_lib : lookup_config:Copybook.TYPES.lookup_config -> ?fromfile:string -> ?libname:Copybook.TYPES.fileloc -> Copybook.TYPES.fileloc -> (string, Copybook.TYPES.lookup_error) Stdlib.result;
}
