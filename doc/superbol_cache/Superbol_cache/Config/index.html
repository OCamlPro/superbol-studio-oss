Module Superbol_cache.Config

val pruning_policy : ?cache_max_age:float -> ?cache_max_size:int64 -> unit -> Types.prune_policy

create ~cache_max_age ~cache_max_size () gathers the cache settings.

cache_max_age (in seconds) and cache_max_size (in bytes) bound what a project's cache keeps; with neither, it grows without limit.
