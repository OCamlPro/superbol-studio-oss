What TYPES.store.stat reports about an item.

type store = {
  1. read : string -> (string option, string) Stdlib.result;
    (*

    read path is Ok None if no item exists at path.

    *)
  2. publish : path:string -> contents:string -> replace:bool -> (unit, string) Stdlib.result;
    (*

    publish ~path ~contents ~replace makes contents appear at path, creating intermediate directories. A reader sees either nothing there or the whole of contents, never a prefix.

    When replace is false, an item already at path is left alone and the call still succeeds: that is what two instances computing the same value should do to each other. When it is true, contents takes the place of whatever was there, last writer winning.

    *)
  3. list : string -> (string list, string) Stdlib.result;
    (*

    list dir is the names of the items directly under dir, without their directory part. An absent directory lists as Ok [].

    *)
  4. stat : string -> (item_info option, string) Stdlib.result;
    (*

    stat path is Ok None unless path holds an item, so that the pruner can tell one from a directory to descend into.

    *)
  5. touch : string -> unit;
    (*

    touch path records that path has just been hit. Best-effort.

    *)
  6. remove : string -> unit;
    (*

    remove path reclaims path. Best-effort.

    *)
  7. with_lock : 'a. string -> f:(unit -> 'a) -> ('a option, string) Stdlib.result;
    (*

    with_lock path ~f runs f with path held exclusively, and is Ok None if the lock is held elsewhere. Implementations are expected to create the lock with O_CREAT | O_EXCL rather than to rely on flock, whose semantics over NFS and SMB cannot be counted on, and to break a lock left behind by a process that died.

    *)
  8. now : unit -> float;
    (*

    Seconds since the epoch.

    *)
}
